R:@app.route("/login", methods=["GET", "POST"])
def login():
    error = None
    last_query = None
    instance_results = None

    instance_param = request.args.get("instance")

    if instance_param:
        query = "SELECT id, name FROM tinstance WHERE name = ?"

        try:
            conn = sqlite3.connect(DB_FILE)
            cursor = conn.cursor()
            cursor.execute(query, (instance_param,))
            instance_results = cursor.fetchall()
            conn.close()

        except sqlite3.Error as e:
            error = f"Erro de sintaxe no SQL (GET instance): {e}"

    if request.method == "POST":
        username = request.form.get("username", "")
        password_raw = request.form.get("password", "")

        password_sha1 = sha1_hash(password_raw)

        query = "SELECT * FROM tperson WHERE login = ? AND password = ?"

        try:
            conn = sqlite3.connect(DB_FILE)
            cursor = conn.cursor()
            cursor.execute(query, (username, password_sha1))
            user = cursor.fetchone()
            conn.close()

            if user:
                session["username"] = user[1]
                session["role"] = user[4]
                return redirect(url_for("dashboard"))
            else:
                error = "Usuário ou senha incorretos."

        except sqlite3.Error as e:
            error = f"Erro de sintaxe no SQL (POST login): {e}"

    return render_template_string(
        LOGIN_TEMPLATE,
        error=error,
        last_query=last_query,
        instance_results=instance_results
    )


@app.route("/produto", methods=["GET"])
def produto():
    error = None
    results = None

    nome_param = request.args.get("nome", "")

    query = "SELECT id, name FROM produtos_financeiros WHERE name = ?"

    try:
        conn = sqlite3.connect(DB_FILE)
        cursor = conn.cursor()
        cursor.execute(query, (nome_param,))
        results = cursor.fetchall()
        conn.close()

    except sqlite3.Error as e:
        error = f"Erro SQL: {e}"

    return render_template_string(
        PRODUTO_TEMPLATE,
        results=results,
        last_query=query,
        error=error
    )
